  function startDrag(e){
    const el=e.currentTarget;if(el.dataset.locked==='1'||el.classList.contains('editing'))return;
    e.preventDefault();e.stopPropagation();selectElement(el.dataset.id);
    const startX=e.clientX,startY=e.clientY,left=parseFloat(el.style.left)||0,top=parseFloat(el.style.top)||0,localScale=scale;
    el.setPointerCapture?.(e.pointerId);
    const move=ev=>{
      const dx=(ev.clientX-startX)/localScale,dy=(ev.clientY-startY)/localScale;
      el.style.left=`${Math.round(left+dx)}px`;el.style.top=`${Math.round(top+dy)}px`;
      if(selectedId===el.dataset.id){document.getElementById('posX').value=Math.round(left+dx);document.getElementById('posY').value=Math.round(top+dy)}
    };
    const up=()=>{el.removeEventListener('pointermove',move);el.removeEventListener('pointerup',up);el.removeEventListener('pointercancel',up);pushHistory()};
    el.addEventListener('pointermove',move);el.addEventListener('pointerup',up);el.addEventListener('pointercancel',up);
  }
