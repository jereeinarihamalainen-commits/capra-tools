(0,0,0,0)');
  ctx.fillStyle=g; ctx.fillRect(0,0,W,H);
  // subtle vignette
  const vg=ctx.createRadialGradient(W*.5,H*.47,300,W*.5,H*.47,900);
  vg.addColorStop(0,'rgba(0,0,0,0)');
  vg.addColorStop(1,'rgba(0,0,0,.18)');
  ctx.fillStyle=vg;ctx.fillRect(0,0,W,H);
}
function lineWrap(text,maxWidth,font){
  ctx.font=font;
  const paras=String(text||'').split('\n');
  const lines=[];
  paras.forEach((p,pi)=>{
    if(p===''){lines.push('');return;}
    const words=p.split(/\s+/);
    let line='';
    for(const word of words){
      const test=line?line+' '+word:word;
      if(ctx.measureText(test).width>maxWidth && line){
        lines.push(line); line=word;
      }else line=test;
    }
    lines.push(line);
  });
  return lines;
}
function drawTextBlock(text,x,y,maxW,size,lh,color,fontFamily,weight){
  const font=`${weight} ${size}px "${fontFamily}"`;
  const lines=lineWrap(text,maxW,font);
  ctx.font=font; ctx.fillStyle=color;ctx.textBaseline='top';ctx.textAlign='left';
  let yy=y, widest=0;
  for(const line of lines){
    if(line!=='') ctx.fillText(line,x,yy);
    widest=Math.max(widest,ctx.measureText(line).width);
    yy += size*lh;
  }
  return {x,y,w:Math.min(maxW,Math.max(widest,20)),h:Math.max(size,lines.length*size*lh)};
}
function drawLogo(s){
  if(!logoImg.complete || !logoImg.naturalWidth) return null;
  const w=s.logoSize;
  const h=w*(logoImg.naturalHeight/logoImg.naturalWidth);
  const x=s.logoX-w/2, y=s.logoY-h/2;
  ctx.save();ctx.globalAlpha=s.logoOpacity;ctx.drawImage(logoImg,x,y,w,h);ctx.restore();
  return {x,y,w,h};
}
function render(){
  const s=slides[current];
  ctx.clearRect(0,0,W,H);
  ctx.fillStyle='#050606';ctx.fillRect(0,0,W,H);
  drawCover(getImg(s.image),s);
  drawOverlays(s);

  hitboxes={};
  if(s.accentW>0){
    ctx.fillStyle=s.accentColor;
    ctx.fillRect(s.accentX,s.accentY,s.accentW,s.accentH);
    hitboxes.accent={x:s.accentX-12,y:s.accentY-12,w:s.accentW+24,h:s.accentH+24};
  }
  if(s.title.trim()) hitboxes.title=drawTextBlock(s.title,s.titleX,s.titleY,s.titleW,s.titleSize,s.titleLH,s.titleColor,s.fontFamily,s.titleWeight);
  if(s.body.trim()) hitboxes.body=drawTextBlock(s.body,s.bodyX,s.bodyY,s.bodyW,s.bodySize,s.bodyLH,s.bodyColor,s.fontFamily,s.bodyWeight);

  if(s.showPage){
    ctx.font='400 28px Georgia';ctx.fillStyle='rgba(248,246,240,.96)';ctx.textAlign='right';ctx.textBaseline='top';
    ctx.fillText(`${current+1}/5`,1008,52);
  }
  hitboxes.logo=drawLogo(s);
}
logoImg.onload=render;

function pointIn(b,x,y){return b&&x>=b.x&&x<=b.x+b.w&&y>=b.y&&y<=b.y+b.h}
function canvasPoint(e){
  const r=canvas.getBoundingClientRect();
  const p=e.touches?e.touches[0]:e;
  return {x:(p.clientX-r.left)*W/r.width,y:(p.clientY-r.top)*H/r.height};
}
canvas.addEventListener('pointerdown',e=>{
  const p=canvasPoint(e), s=slides[current];
  for(const key of ['title','body','logo','accent']){
    if(pointIn(hitboxes[key],p.x,p.y)){
      drag={key,start:p,orig:{}};
      if(key==='title') drag.orig={x:s.titleX,y:s.titleY};
      if(key==='body') drag.orig={x:s.bodyX,y:s.bodyY};
      if(key==='logo') drag.orig={x:s.logoX,y:s.logoY};
      if(key==='accent') drag.orig={x:s.accentX,y:s.accentY};
      canvas.setPointerCapture(e.pointerId);canvas.classList.add('dragging');break;
    }
  }
});
canvas.addEventListener('pointermove',e=>{
  if(!drag)return;
  const p=canvasPoint(e),s=slides[current],dx=p.x-drag.start.x,dy=p.y-drag.start.y;
  if(drag.key==='title'){s.titleX=drag.orig.x+dx;s.titleY=drag.orig.y+dy}
  if(drag.key==='body'){s.bodyX=drag.orig.x+dx;s.bodyY=drag.orig.y+dy}
  if(drag.key==='logo'){s.logoX=drag.orig.x+dx;s.logoY=drag.orig.y+dy}
  if(drag.key==='accent'){s.accentX=drag.orig.x+dx;s.accentY=drag.orig.y+dy}
  render();
});
function endDrag(){drag=null;canvas.classList.remove('dragging');}
canvas.addEventListener('pointerup',endDrag);canvas.addEventListener('pointercancel',endDrag);
