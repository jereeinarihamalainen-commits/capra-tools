(state.theme) themeSelect.value = state.theme;
          if (state.accent) accentColor.value = state.accent;
          if (state.typeScale) typeScale.value = state.typeScale;
          if (state.previewZoom) previewZoom.value = state.previewZoom;
          if (state.photoPosition !== undefined) photoPosition.value = state.photoPosition;
          if (state.photoData) photoData = state.photoData;
          if (state.heroBackgroundDarkness !== undefined) heroBackgroundDarkness.value = state.heroBackgroundDarkness;
          if (state.heroBackgroundPosition !== undefined) heroBackgroundPosition.value = state.heroBackgroundPosition;
          if (state.heroBackgroundData) heroBackgroundData = state.heroBackgroundData;
          if (state.coachPhotoPosition !== undefined) coachPhotoPosition.value = state.coachPhotoPosition;
          if (state.coachPhotoData) coachPhotoData = state.coachPhotoData;
          if (state.extras) extraElements.innerHTML = state.extras;
        } catch (error) {
          updateStatus("Luonnosta ei voitu avata");
        }
      }

      function resizeImage(file) {
        return new Promise((resolve, reject) => {
          const reader = new FileReader();
          reader.onerror = reject;
          reader.onload = () => {
            const image = new Image();
            image.onerror = reject;
            image.onload = () => {
              const max = 1600;
              const scale = Math.min(1, max / Math.max(image.width, image.height));
              const canvas = document.createElement("canvas");
              canvas.width = Math.round(image.width * scale);
              canvas.height = Math.round(image.height * scale);
              canvas.getContext("2d").drawImage(image, 0, 0, canvas.width, canvas.height);
              resolve(canvas.toDataURL("image/jpeg", 0.84));
            };
            image.src = reader.result;
          };
          reader.readAsDataURL(file);
        });
      }

      function syncFormAttributes() {
        document.querySelectorAll("select option").forEach((option) => option.removeAttribute("selected"));
        themeSelect.querySelector(`option[value="${themeSelect.value}"]`)?.setAttribute("selected", "selected");
        toggleInputs().forEach((input) => {
          if (input.checked) input.setAttribute("checked", "checked");
          else input.removeAttribute("checked");
        });
        accentColor.setAttribute("value", accentColor.value);
        typeScale.setAttribute("value", typeScale.value);
        previewZoom.setAttribute("value", previewZoom.value);
        photoPosition.setAttribute("value", photoPosition.value);
        heroBackgroundDarkness.setAttribute("value", heroBackgroundDarkness.value);
        heroBackgroundPosition.setAttribute("value", heroBackgroundPosition.value);
        coachPhotoPosition.setAttribute("value", coachPhotoPosition.value);
      }

      function downloadCurrentHtml() {
        syncFormAttributes();
        const source = "<!doctype html>\n" + document.documentElement.outerHTML;
        const blob = new Blob([source], { type: "text/html;charset=utf-8" });
        const url = URL.createObjectURL(blob);
        const link = document.createElement("a");
        link.href = url;
        link.download = "capra-junioriesite-muokattu.html";
        document.body.appendChild(link);
        link.click();
        link.remove();
        setTimeout(() => URL.revokeObjectURL(url), 1000);
      }
