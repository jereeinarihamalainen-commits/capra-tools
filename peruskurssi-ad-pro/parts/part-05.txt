function wrapLines(ctx,content,maxWidth){
 const paras=String(content??"").split("\n"),lines=[];
 paras.forEach((p,pi)=>{if(p===""){lines.push("");return}const words=p.split(/\s+/);let line="";
  words.forEach(w=>{const test=line?line+" "+w:w;if(ctx.measureText(test).width<=maxWidth||!line)line=test;else{lines.push(line);line=w}});if(line)lines.push(line);
 });
 return lines;
}
function drawSpaced(ctx,str,x,y,spacing,align,width){
 if(!spacing){ctx.textAlign=align;ctx.fillText(str,align==="center"?x+width/2:align==="right"?x+width:x,y);return}
 const chars=[...str],total=chars.reduce((n,ch)=>n+ctx.measureText(ch).width,0)+spacing*Math.max(0,chars.length-1);
 let sx=align==="center"?x+(width-total)/2:align==="right"?x+width-total:x;ctx.textAlign="left";
 chars.forEach(ch=>{ctx.fillText(ch,sx,y);sx+=ctx.measureText(ch).width+spacing})
}

function roundRect(ctx,x,y,w,h,r){r=Math.max(0,Math.min(r,w/2,h/2));ctx.beginPath();ctx.moveTo(x+r,y);ctx.arcTo(x+w,y,x+w,y+h,r);ctx.arcTo(x+w,y+h,x,y+h,r);ctx.arcTo(x,y+h,x,y,r);ctx.arcTo(x,y,x+w,y,r);ctx.closePath()}
